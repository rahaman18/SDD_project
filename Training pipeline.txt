# preprocess.py
import os
import random
from sklearn.model_selection import StratifiedKFold
from audiomentations import Compose, AddGaussianNoise, TimeStretch, PitchShift


def preprocess_dataset(data_path, augment=True, k_fold=5):
    # Simulated data loader; each sample has 'filepath' and 'label'
    data = [{'filepath': os.path.join(data_path, f), 'label': 0 if 'healthy' in f else 1}
            for f in os.listdir(data_path) if f.endswith('.wav')]

    skf = StratifiedKFold(n_splits=k_fold, shuffle=True, random_state=42)
    X = [d['filepath'] for d in data]
    y = [d['label'] for d in data]

    splits = list(skf.split(X, y))
    train_idx, test_idx = splits[0]

    train_data = [data[i] for i in train_idx]
    test_data = [data[i] for i in test_idx]

    if augment:
        augmenter = Compose([
            AddGaussianNoise(min_amplitude=0.001, max_amplitude=0.015, p=0.5),
            TimeStretch(min_rate=0.8, max_rate=1.25, p=0.5),
            PitchShift(min_semitones=-4, max_semitones=4, p=0.5)
        ])
        for d in train_data:
            d['augmented'] = True

    return train_data, None, test_data


# xgboost_classifier.py
import xgboost as xgb

class XGBoostClassifier:
    def __init__(self, params):
        self.model = xgb.XGBClassifier(**params)

    def train(self, X, y):
        self.model.fit(X, y)

    def predict(self, X):
        return self.model.predict(X)


# bohb_tuner.py
from sklearn.model_selection import train_test_split
from hyperopt import fmin, tpe, hp, Trials


def tune_hyperparameters(X, y):
    def objective(params):
        model = xgb.XGBClassifier(**params)
        X_train, X_val, y_train, y_val = train_test_split(X, y, test_size=0.2)
        model.fit(X_train, y_train)
        acc = model.score(X_val, y_val)
        return -acc

    space = {
        'max_depth': hp.choice('max_depth', [3, 4, 5, 6]),
        'learning_rate': hp.uniform('learning_rate', 0.01, 0.3),
        'n_estimators': hp.choice('n_estimators', [100, 200, 300]),
    }

    trials = Trials()
    best = fmin(fn=objective, space=space, algo=tpe.suggest, max_evals=30, trials=trials)
    return best

# metrics.py
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score

def evaluate_model(y_true, y_pred):
    print("Accuracy:", accuracy_score(y_true, y_pred))
    print("Precision:", precision_score(y_true, y_pred))
    print("Recall:", recall_score(y_true, y_pred))
    print("F1 Score:", f1_score(y_true, y_pred))
